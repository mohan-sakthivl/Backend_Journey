choice = int(input("Enter your choice 1-4: "))
match choice:
    case 1:
        print("Pizza")
    case 2:
        print("Burger")
    case 3:
        print("Pasta")
    case 4:
        print("Fried Rice")
    case _:
        print("Invalid choice")
